import 'package:flutter_sms/features/mission/domain/entities/mission_status.dart';

/// One mission card on the home page.
/// TODO: add fromJson() when the real API is connected.
class HomeMission {
  const HomeMission({
    required this.id,
    required this.code,
    required this.title,
    required this.location,
    required this.start,
    required this.end,
    required this.type,
    required this.transport,
    required this.status,
    required this.amount,
    this.note,
    this.requesterName = '',
    this.department = '',
    this.isMine = false,
  });

  final String id;
  final String code; // e.g. MSN-2026-0128
  final String title;
  final String location;
  final DateTime start;
  final DateTime end;
  final String type; // e.g. Individual
  final String transport; // e.g. Company Vehicle
  final MissionStatus status;
  final double amount;
  final String? note;
  final String requesterName;
  final String department;
  final bool isMine;

  static const _months = [
    'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
    'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec',
  ];

  int get days => end.difference(start).inDays + 1;

  String get amountLabel => '\$${amount.toStringAsFixed(2)}';

  String get dateLabel {
    String two(int n) => n.toString().padLeft(2, '0');
    final m1 = _months[start.month - 1];
    final m2 = _months[end.month - 1];
    final String range;
    if (start.year == end.year &&
        start.month == end.month &&
        start.day == end.day) {
      range = '${two(start.day)} $m1 ${start.year}';
    } else if (start.year == end.year && start.month == end.month) {
      range = '${two(start.day)} \u2013 ${two(end.day)} $m1 ${end.year}';
    } else {
      range = '${two(start.day)} $m1 \u2013 ${two(end.day)} $m2 ${end.year}';
    }
    final unit = days == 1 ? 'day' : 'days';
    return '$range \u00B7 $days $unit';
  }
}

import 'package:flutter/material.dart';

/// Mirrors the backend enum: com.onemore.mission.mission.domain.MissionStatus
enum MissionStatus {
  draft('DRAFT', 'Draft', Color(0xFF6B7280), Color(0xFFF1F2F4)),
  submitted('SUBMITTED', 'Submitted', Color(0xFF1D4ED8), Color(0xFFE6EEFC)),
  fmReview('FM_REVIEW', 'FM Review', Color(0xFFB45309), Color(0xFFFEF3C7)),
  hrbpReview('HRBP_REVIEW', 'HRBP Review', Color(0xFFB45309), Color(0xFFFEF3C7)),
  financeReview('FINANCE_REVIEW', 'Finance Review', Color(0xFFB45309), Color(0xFFFEF3C7)),
  bizopsReview('BIZOPS_REVIEW', 'BizOps Review', Color(0xFFB45309), Color(0xFFFEF3C7)),
  executiveReview('EXECUTIVE_REVIEW', 'Executive Review', Color(0xFFB45309), Color(0xFFFEF3C7)),
  approved('APPROVED', 'Approved', Color(0xFF15803D), Color(0xFFDCFCE7)),
  rejected('REJECTED', 'Rejected', Color(0xFFB91C1C), Color(0xFFFEE2E2)),
  cancelled('CANCELLED', 'Cancelled', Color(0xFF4B5563), Color(0xFFE5E7EB)),
  reportSubmitted('REPORT_SUBMITTED', 'Report Submitted', Color(0xFF0F766E), Color(0xFFCCFBF1)),
  settled('SETTLED', 'Settled', Color(0xFF6D28D9), Color(0xFFEDE9FE));

  const MissionStatus(this.code, this.label, this.color, this.background);

  /// Value sent by the backend, e.g. FM_REVIEW.
  final String code;
  final String label;
  final Color color;
  final Color background;

  static MissionStatus? tryParse(String? raw) {
    if (raw == null) return null;
    final value = raw.trim().toUpperCase();
    for (final s in MissionStatus.values) {
      if (s.code == value) return s;
    }
    return null;
  }
}

/// Which statuses are "waiting for me" for each role.
extension UserRoleQueue on UserRole {
  Set<MissionStatus> get queueStatuses {
    switch (this) {
      case UserRole.staff:
        return const <MissionStatus>{};
      case UserRole.functionManager:
        return const {MissionStatus.fmReview};
      case UserRole.hrbp:
        return const {MissionStatus.hrbpReview};
      case UserRole.finance:
        return const {MissionStatus.financeReview, MissionStatus.reportSubmitted};
      case UserRole.bizops:
        return const {MissionStatus.bizopsReview};
      case UserRole.executive:
        return const {MissionStatus.executiveReview};
      case UserRole.admin:
        return const {
          MissionStatus.fmReview,
          MissionStatus.hrbpReview,
          MissionStatus.financeReview,
          MissionStatus.bizopsReview,
          MissionStatus.executiveReview,
        };
    }
  }
}

import 'package:dio/dio.dart';

import '../storage/token_storage.dart';
import 'api_config.dart';

class DioClient {
  final Dio dio;
  final TokenStorage _tokenStorage;

  DioClient()
    : _tokenStorage = TokenStorage(),
      dio = Dio(
        BaseOptions(
          baseUrl: ApiConfig.baseUrl,
          headers: {'Content-Type': 'application/json'},
          connectTimeout: const Duration(seconds: 10),
          receiveTimeout: const Duration(seconds: 10),
        ),
      ) {
    dio.interceptors.add(
      InterceptorsWrapper(
        onRequest: (options, handler) async {
          final token = await _tokenStorage.getToken();

          if (token != null && token.isNotEmpty) {
            options.headers['Authorization'] = 'Bearer $token';
          }

          handler.next(options);
        },
      ),
    );
  }

  Future<void> testConnection() async {
    try {
      final response = await dio.get('/swagger-ui/index.html');

      print('API connected: ${response.statusCode}');
    } catch (e) {
      print('API connection failed: $e');
    }
  }
}

import 'package:flutter_sms/core/storage/session_storage.dart';
import 'package:shared_preferences/shared_preferences.dart';

class TokenStorage {
  static const String _tokenKey = 'auth_token';

  Future<void> saveToken(String token) async {
    final prefs = await SharedPreferences.getInstance();

    await prefs.setString(_tokenKey, token);
  }

  Future<String?> getToken() async {
    final prefs = await SharedPreferences.getInstance();

    return prefs.getString(_tokenKey);
  }

  /// Logout: removes the token and the saved role/name.
  Future<void> clearToken() async {
    final prefs = await SharedPreferences.getInstance();

    await prefs.remove(_tokenKey);
    await SessionStorage().clear();
  }
}

import '../../../core/network/dio_client.dart';
import 'models/create_mission_request.dart';
import 'models/employee.dart';
import 'models/mission_response.dart';

class MissionRepository {
  final DioClient _dioClient;

  MissionRepository(this._dioClient);

  Future<MissionResponse> createMission(CreateMissionRequest request) async {
    final response = await _dioClient.dio.post(
      '/api/missions',
      data: request.toJson(),
    );

    final responseData = response.data as Map<String, dynamic>;

    final data = responseData['data'];

    if (data is Map<String, dynamic>) {
      return MissionResponse.fromJson(data);
    }

    return MissionResponse.fromJson(responseData);
  }

  Future<MissionResponse> getMission(int id) async {
    final response = await _dioClient.dio.get('/api/missions/$id');

    final responseData = response.data as Map<String, dynamic>;

    final data = responseData['data'];

    if (data is Map<String, dynamic>) {
      return MissionResponse.fromJson(data);
    }

    return MissionResponse.fromJson(responseData);
  }

  Future<List<MissionResponse>> getMissions() async {
    final response = await _dioClient.dio.get('/api/missions');

    final responseData = response.data as Map<String, dynamic>;

    final data = responseData['data'];

    if (data is Map<String, dynamic>) {
      final items = data['items'] as List<dynamic>? ?? [];

      return items
          .map((item) => MissionResponse.fromJson(item as Map<String, dynamic>))
          .toList();
    }

    final items = responseData['items'] as List<dynamic>? ?? [];

    return items
        .map((item) => MissionResponse.fromJson(item as Map<String, dynamic>))
        .toList();
  }

  Future<void> deleteMission(int id) async {
    await _dioClient.dio.delete('/api/missions/$id');
  }

  Future<void> updateParticipants(
    int missionId,
    List<int> participantIds,
  ) async {
    await _dioClient.dio.put(
      '/api/missions/$missionId/participants',
      data: {'participantIds': participantIds},
    );
  }

  Future<List<Employee>> getEmployees() async {
    final response = await _dioClient.dio.get('/api/users');

    final responseData = response.data as Map<String, dynamic>;

    final data = responseData['data'];

    if (data is Map<String, dynamic>) {
      final items = data['items'] as List<dynamic>? ?? [];

      return items
          .map((item) => Employee.fromJson(item as Map<String, dynamic>))
          .where((employee) => employee.isActive)
          .toList();
    }

    return [];
  }
}
